Security Contract06 / 12

The Security Contract

capabilities:
  rag: true
  memory: true
  tools: true
  multi_tenant: true
  structured_output: true

security:
  prompt_injection: true
  sensitive_information_disclosure: true
  excessive_agency: true

constraints:
  allowed_methods:
    - GET
    - POST
